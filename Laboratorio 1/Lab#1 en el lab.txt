import java.util.*;

// Clase que representa un nodo del Árbol B
class Nodo {
    int[] claves;
    Nodo[] hijos;
    int numClaves;
    boolean esHoja;
    int orden;

    public Nodo(int orden, boolean esHoja) {
        this.orden = orden;
        this.esHoja = esHoja;
        this.claves = new int[2 * orden - 1];
        this.hijos = new Nodo[2 * orden];
        this.numClaves = 0;
    }

    public void insertarNoLleno(int clave) {
        int i = numClaves - 1;

        if (esHoja) {
            while (i >= 0 && claves[i] > clave) {
                claves[i + 1] = claves[i];
                i--;
            }
            claves[i + 1] = clave;
            numClaves++;
        } else {
            while (i >= 0 && claves[i] > clave) {
                i--;
            }
            i++;

            if (hijos[i].numClaves == 2 * orden - 1) {
                dividirHijo(i, hijos[i]);

                if (claves[i] < clave) {
                    i++;
                }
            }
            hijos[i].insertarNoLleno(clave);
        }
    }

    public void dividirHijo(int indice, Nodo hijoLleno) {
        Nodo nuevoHijo = new Nodo(orden, hijoLleno.esHoja);
        nuevoHijo.numClaves = orden - 1;

        for (int j = 0; j < orden - 1; j++) {
            nuevoHijo.claves[j] = hijoLleno.claves[j + orden];
        }

        if (!hijoLleno.esHoja) {
            for (int j = 0; j < orden; j++) {
                nuevoHijo.hijos[j] = hijoLleno.hijos[j + orden];
            }
        }

        hijoLleno.numClaves = orden - 1;

        for (int j = numClaves; j >= indice + 1; j--) {
            hijos[j + 1] = hijos[j];
        }

        hijos[indice + 1] = nuevoHijo;

        for (int j = numClaves - 1; j >= indice; j--) {
            claves[j + 1] = claves[j];
        }

        claves[indice] = hijoLleno.claves[orden - 1];
        numClaves++;
    }
}

// Clase principal del Árbol B
public class ArbolB {
    private Nodo raiz;
    private int orden;

    public ArbolB(int orden) {
        this.orden = orden;
        this.raiz = null;
    }

    public void insertar(int clave) {
        if (raiz == null) {
            raiz = new Nodo(orden, true);
            raiz.claves[0] = clave;
            raiz.numClaves = 1;
        } else {
            if (raiz.numClaves == 2 * orden - 1) {
                Nodo nuevaRaiz = new Nodo(orden, false);
                nuevaRaiz.hijos[0] = raiz;

                nuevaRaiz.dividirHijo(0, raiz);

                int i = 0;
                if (nuevaRaiz.claves[0] < clave) {
                    i++;
                }
                nuevaRaiz.hijos[i].insertarNoLleno(clave);

                raiz = nuevaRaiz;
            } else {
                raiz.insertarNoLleno(clave);
            }
        }
    }

    // Método para mostrar el árbol por niveles
    public void mostrarPorNiveles() {
        if (raiz == null) {
            System.out.println("El árbol está vacío");
            return;
        }

        Queue<Nodo> cola = new LinkedList<>();
        Queue<Integer> niveles = new LinkedList<>();

        cola.add(raiz);
        niveles.add(0);

        int nivelActual = -1;
        while (!cola.isEmpty()) {
            Nodo nodo = cola.poll();
            int nivel = niveles.poll();

            if (nivel != nivelActual) {
                if (nivelActual != -1) System.out.println();
                System.out.print("Nivel " + nivel + ": ");
                nivelActual = nivel;
            }

            System.out.print("(");
            for (int i = 0; i < nodo.numClaves; i++) {
                System.out.print(nodo.claves[i]);
                if (i < nodo.numClaves - 1) System.out.print(", ");
            }
            System.out.print(") ");

            if (!nodo.esHoja) {
                for (int i = 0; i <= nodo.numClaves; i++) {
                    if (nodo.hijos[i] != null) {
                        cola.add(nodo.hijos[i]);
                        niveles.add(nivel + 1);
                    }
                }
            }
        }
        System.out.println();
    }

    // Método principal
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        ArbolB arbol = new ArbolB(2); // orden 2, puedes cambiarlo si deseas

        while (true) {
            System.out.print("Ingrese un número para insertar (o escriba 'salir' para terminar): ");
            String entrada = scanner.nextLine().trim();

            if (entrada.equalsIgnoreCase("salir")) {
                break;
            }

            try {
                int clave = Integer.parseInt(entrada);
                arbol.insertar(clave);
                arbol.mostrarPorNiveles();
            } catch (NumberFormatException e) {
                System.out.println("Entrada no válida. Intente con un número entero.");
            }
        }

        System.out.println("Programa terminado.");
    }
}
